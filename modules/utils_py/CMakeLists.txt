
# Python 扩展模块：绑定 Python 嵌入运行时、pybind11、General 和 Math。
# Python 解释器与扩展运行时只在显式启用 Utils_py 时查找。
find_package(Python3 3.12 REQUIRED COMPONENTS Interpreter Development.Embed)
if(Python3_LIBRARIES)
    list(GET Python3_LIBRARIES 0 SINDRE_UTILS_PY_PYTHON_LIBRARY)
elseif(Python3_LIBRARY)
    set(SINDRE_UTILS_PY_PYTHON_LIBRARY "${Python3_LIBRARY}")
endif()
if(NOT SINDRE_UTILS_PY_PYTHON_LIBRARY)
    find_library(SINDRE_UTILS_PY_PYTHON_LIBRARY
        NAMES "python${Python3_VERSION_MAJOR}${Python3_VERSION_MINOR}"
        HINTS "${Python3_LIBRARY_DIRS}" "${Python3_ROOT_DIR}"
        PATH_SUFFIXES lib libs)
endif()
if(NOT SINDRE_UTILS_PY_PYTHON_LIBRARY)
    message(FATAL_ERROR "Python development import library was not found")
endif()
sindre_3rdparty_setup_pybind11()
# The installed package config is generated by the repository root.  Export
# the bundled-provider decision to that directory instead of keeping it
# local to utils_py.
set(SINDRE_PYBIND11_BUNDLED ON PARENT_SCOPE)

add_library(sindre_utils_py SHARED src/python.cpp)
add_library(sindre::utils_py ALIAS sindre_utils_py)
target_compile_features(sindre_utils_py PUBLIC cxx_std_17)
target_include_directories(sindre_utils_py
    PUBLIC
        $<BUILD_INTERFACE:${SINDRE_SOURCE_DIR}/include>
        $<BUILD_INTERFACE:${Python3_INCLUDE_DIRS}>
        $<INSTALL_INTERFACE:include>)
target_compile_definitions(sindre_utils_py
    PUBLIC SINDRE_WITH_UTILS_PY=1 SINDRE_UTILS_PY_SHARED=1
    PRIVATE SINDRE_UTILS_PY_BUILD=1)
target_link_libraries(sindre_utils_py
    PUBLIC sindre::general sindre::math sindre::pybind11_embed
        $<BUILD_INTERFACE:${SINDRE_UTILS_PY_PYTHON_LIBRARY}>
        $<INSTALL_INTERFACE:Python3::Python>)
set_target_properties(sindre_utils_py PROPERTIES EXPORT_NAME utils_py)
set(_sindre_utils_py_runtime_files)
# A uv-managed virtual environment keeps python.exe in .venv\Scripts while
# the embedded interpreter home belongs to the base CPython installation.
# Resolve that home on every platform; Windows additionally uses it to locate
# the interpreter DLL that must sit beside embedded executables.
execute_process(
    COMMAND "${Python3_EXECUTABLE}" -c "import sys; print(sys.base_prefix)"
    OUTPUT_VARIABLE _sindre_utils_py_python_base_prefix
    OUTPUT_STRIP_TRAILING_WHITESPACE
    COMMAND_ERROR_IS_FATAL ANY)
file(TO_CMAKE_PATH "${_sindre_utils_py_python_base_prefix}"
    _sindre_utils_py_python_base_prefix)
if(WIN32)
    # Looking only next to Python3_EXECUTABLE leaves embedded processes with
    # STATUS_DLL_NOT_FOUND (0xc0000135) when Python came from uv.
    find_file(_sindre_utils_py_python_runtime
        NAMES "python${Python3_VERSION_MAJOR}${Python3_VERSION_MINOR}.dll"
        HINTS "${_sindre_utils_py_python_base_prefix}"
              "${Python3_ROOT_DIR}" "${Python3_EXECUTABLE}"
        PATH_SUFFIXES . DLLs bin NO_DEFAULT_PATH)
    if(_sindre_utils_py_python_runtime)
        list(APPEND _sindre_utils_py_runtime_files
            "${_sindre_utils_py_python_runtime}")
    endif()
    if(DEFINED SINDRE_THIRD_GENERAL_CS_STRING_RUNTIME AND
       EXISTS "${SINDRE_THIRD_GENERAL_CS_STRING_RUNTIME}")
        list(APPEND _sindre_utils_py_runtime_files
            "${SINDRE_THIRD_GENERAL_CS_STRING_RUNTIME}")
    endif()
endif()
sindre_copy_runtime_files(sindre_utils_py ${_sindre_utils_py_runtime_files})
if(WIN32)
    foreach(_sindre_utils_py_runtime_file IN LISTS _sindre_utils_py_runtime_files)
        if(EXISTS "${_sindre_utils_py_runtime_file}")
            install(FILES "${_sindre_utils_py_runtime_file}"
                DESTINATION ${CMAKE_INSTALL_BINDIR})
        endif()
    endforeach()
endif()

if(SINDRE_BUILD_TESTS)
    set(_sindre_utils_py_test_python_home
        "${_sindre_utils_py_python_base_prefix}")
    file(TO_CMAKE_PATH "${Python3_EXECUTABLE}"
        _sindre_utils_py_test_python_executable)
    add_executable(sindre_utils_py_tests tests/python_runtime_tests.cpp)
    target_compile_definitions(sindre_utils_py_tests PRIVATE
        SINDRE_TEST_PYTHON_EXECUTABLE="${_sindre_utils_py_test_python_executable}"
        SINDRE_TEST_PYTHON_HOME="${_sindre_utils_py_test_python_home}")
    target_link_libraries(sindre_utils_py_tests PRIVATE sindre::utils_py)
    sindre_copy_runtime_files(sindre_utils_py_tests ${_sindre_utils_py_runtime_files})
    add_test(NAME sindre.utils_py.runtime COMMAND sindre_utils_py_tests)

    if(SINDRE_WITH_UTILS_3D)
        add_executable(sindre_utils_py_numpy_tests tests/python_numpy_tests.cpp)
        target_compile_definitions(sindre_utils_py_numpy_tests PRIVATE
            SINDRE_TEST_PYTHON_EXECUTABLE="${_sindre_utils_py_test_python_executable}"
            SINDRE_TEST_PYTHON_HOME="${_sindre_utils_py_test_python_home}")
        target_link_libraries(sindre_utils_py_numpy_tests
            PRIVATE sindre::utils_py sindre::utils_3d)
        sindre_copy_runtime_files(sindre_utils_py_numpy_tests
            ${_sindre_utils_py_runtime_files})
        add_test(NAME sindre.utils_py.numpy COMMAND sindre_utils_py_numpy_tests)
    endif()
endif()
