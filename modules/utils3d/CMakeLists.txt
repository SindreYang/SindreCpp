set(_sindrecpp_utils3d_include "${CMAKE_CURRENT_SOURCE_DIR}/include")
include("${SINDRECPP_THIRDS_DIR}/utils3d/Dependencies.cmake")
set(_sindrecpp_vtk_components CommonCore CommonDataModel CommonTransforms
    FiltersCore FiltersGeneral FiltersGeometry FiltersModeling IOGeometry IOLegacy IOPLY IOXML)
if(SINDRECPP_UTILS3D_SHOW OR SINDRECPP_UTILS3D_VTK_DATA)
    enable_language(C)
endif()
if(SINDRECPP_UTILS3D_SHOW)
    list(APPEND _sindrecpp_vtk_components RenderingCore RenderingOpenGL2 RenderingFreeType
        RenderingAnnotation InteractionStyle IOImage)
endif()
if(SINDRECPP_UTILS3D_VTK_DATA)
    list(APPEND _sindrecpp_vtk_components FiltersFlowPaths FiltersSources ImagingCore ImagingGeneral ImagingMath ImagingMorphological)
    if(SINDRECPP_UTILS3D_SHOW)
        list(APPEND _sindrecpp_vtk_components RenderingVolumeOpenGL2 RenderingContextOpenGL2
            RenderingContext2D ChartsCore ViewsContext2D)
    endif()
endif()
find_package(VTK ${SINDRECPP_THIRD_UTILS3D_VTK_MIN_VERSION}
    REQUIRED COMPONENTS ${_sindrecpp_vtk_components})
find_package(Eigen3 CONFIG QUIET NO_MODULE)
if(NOT TARGET Eigen3::Eigen)
    sindrecpp_thirds_declare_git(eigen
        "${SINDRECPP_THIRD_UTILS3D_EIGEN_REPOSITORY}"
        "${SINDRECPP_THIRD_UTILS3D_EIGEN_TAG}")
    sindrecpp_make_available(eigen EIGEN_BUILD_DOC=OFF EIGEN_BUILD_PKGCONFIG=OFF)
endif()

sindrecpp_add_interface_target(Utils3d "${_sindrecpp_utils3d_include}")
target_compile_definitions(SindreCpp_Utils3d INTERFACE SINDRECPP_WITH_UTILS3D=1)
target_link_libraries(SindreCpp_Utils3d INTERFACE Eigen3::Eigen ${VTK_LIBRARIES})
if(SINDRECPP_UTILS3D_NATIVE_ARCH)
    add_library(sindrecpp_utils3d_native_arch INTERFACE)
    if(MSVC)
        target_compile_options(sindrecpp_utils3d_native_arch INTERFACE /arch:AVX2)
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang|AppleClang")
        target_compile_options(sindrecpp_utils3d_native_arch INTERFACE -march=native)
    endif()
    target_link_libraries(SindreCpp_Utils3d INTERFACE sindrecpp_utils3d_native_arch)
    install(TARGETS sindrecpp_utils3d_native_arch EXPORT SindreCppTargets)
endif()

string(TOUPPER "${SINDRECPP_UTILS3D_BLAS_BACKEND}" _sindrecpp_blas_backend)
if(NOT _sindrecpp_blas_backend MATCHES "^(AUTO|EIGEN|BLAS)$")
    message(FATAL_ERROR "SINDRECPP_UTILS3D_BLAS_BACKEND must be AUTO, EIGEN, or BLAS")
endif()
if(NOT TARGET BLAS::BLAS AND NOT _sindrecpp_blas_backend STREQUAL "EIGEN")
    if(_sindrecpp_blas_backend STREQUAL "BLAS")
        find_package(BLAS REQUIRED)
    else()
        find_package(BLAS QUIET)
    endif()
endif()
if(TARGET BLAS::BLAS AND NOT _sindrecpp_blas_backend STREQUAL "EIGEN")
    target_link_libraries(SindreCpp_Utils3d INTERFACE BLAS::BLAS)
    target_compile_definitions(SindreCpp_Utils3d INTERFACE EIGEN_USE_BLAS=1)
endif()

foreach(backend IN ITEMS MESHLIB CGAL OPEN3D IGL VCG)
    if(SINDRECPP_UTILS3D_${backend})
        target_compile_definitions(SindreCpp_Utils3d INTERFACE SINDRECPP_UTILS3D_${backend}=1)
    endif()
endforeach()
if(SINDRECPP_UTILS3D_SHOW)
    target_compile_definitions(SindreCpp_Utils3d INTERFACE SINDRECPP_UTILS3D_SHOW=1)
endif()
if(SINDRECPP_UTILS3D_VTK_DATA)
    target_compile_definitions(SindreCpp_Utils3d INTERFACE SINDRECPP_UTILS3D_VTK_DATA=1)
endif()
if(SINDRECPP_UTILS3D_CGAL)
    find_package(CGAL REQUIRED)
    target_link_libraries(SindreCpp_Utils3d INTERFACE CGAL::CGAL)
endif()
if(SINDRECPP_UTILS3D_OPEN3D)
    find_package(Open3D CONFIG REQUIRED)
    target_link_libraries(SindreCpp_Utils3d INTERFACE Open3D::Open3D)
endif()
if(SINDRECPP_UTILS3D_MESHLIB)
    find_package(meshlib CONFIG REQUIRED)
    if(TARGET MeshLib::MRMesh)
        target_link_libraries(SindreCpp_Utils3d INTERFACE MeshLib::MRMesh)
    elseif(TARGET MRMesh)
        target_link_libraries(SindreCpp_Utils3d INTERFACE MRMesh)
    else()
        message(FATAL_ERROR "MRMesh SDK must export MeshLib::MRMesh or MRMesh")
    endif()
endif()
if(SINDRECPP_UTILS3D_IGL)
    find_path(_sindrecpp_igl_include igl/harmonic.h HINTS ${SINDRECPP_IGL_ROOT}/include REQUIRED)
    target_include_directories(SindreCpp_Utils3d SYSTEM INTERFACE ${_sindrecpp_igl_include})
endif()
if(SINDRECPP_UTILS3D_VCG)
    find_path(_sindrecpp_vcg_include vcg/complex/complex.h HINTS ${SINDRECPP_VCG_ROOT} REQUIRED)
    target_include_directories(SindreCpp_Utils3d SYSTEM INTERFACE ${_sindrecpp_vcg_include})
endif()

if(WIN32 AND VTK_DIR)
    get_filename_component(_sindrecpp_vtk_root "${VTK_DIR}" DIRECTORY)
    set(_sindrecpp_vtk_runtime_dir "${_sindrecpp_vtk_root}/content/bin")
    if(EXISTS "${_sindrecpp_vtk_runtime_dir}")
        set_property(TARGET SindreCpp_Utils3d PROPERTY SINDRECPP_RUNTIME_DIRS
            "${_sindrecpp_vtk_runtime_dir}")
    endif()
endif()

if(SINDRECPP_BUILD_TESTS)
    add_executable(sindrecpp_utils3d_tests tests/utils3d_tests.cpp)
    target_link_libraries(sindrecpp_utils3d_tests PRIVATE SindreCpp::Utils3d)
    sindrecpp_copy_target_runtime_dlls(sindrecpp_utils3d_tests)
    add_test(NAME sindrecpp.utils3d COMMAND sindrecpp_utils3d_tests)
    add_executable(sindrecpp_mesh_tests tests/mesh_tests.cpp)
    target_link_libraries(sindrecpp_mesh_tests PRIVATE SindreCpp::Utils3d)
    sindrecpp_copy_target_runtime_dlls(sindrecpp_mesh_tests)
    add_test(NAME sindrecpp.mesh COMMAND sindrecpp_mesh_tests)
    if(SINDRECPP_UTILS3D_VTK_DATA)
        add_executable(sindrecpp_vtk_data_tests tests/vtk_data_tests.cpp)
        target_link_libraries(sindrecpp_vtk_data_tests PRIVATE SindreCpp::Utils3d)
        sindrecpp_copy_target_runtime_dlls(sindrecpp_vtk_data_tests)
        add_test(NAME sindrecpp.vtk_data COMMAND sindrecpp_vtk_data_tests)
    endif()
    if(SINDRECPP_UTILS3D_SHOW)
        add_executable(sindrecpp_show_mesh_tests tests/show_mesh_tests.cpp)
        target_link_libraries(sindrecpp_show_mesh_tests PRIVATE SindreCpp::Utils3d)
        sindrecpp_copy_target_runtime_dlls(sindrecpp_show_mesh_tests)
        add_test(NAME sindrecpp.show_mesh COMMAND sindrecpp_show_mesh_tests)
    endif()

    if(WIN32 AND VTK_DIR)
        if(EXISTS "${_sindrecpp_vtk_runtime_dir}")
            set(_sindrecpp_vtk_path_modification
                "PATH=path_list_prepend:${_sindrecpp_vtk_runtime_dir}")
            set(_sindrecpp_vtk_tests sindrecpp.utils3d sindrecpp.mesh)
            foreach(_sindrecpp_vtk_target IN ITEMS sindrecpp_utils3d_tests sindrecpp_mesh_tests)
                if(TARGET ${_sindrecpp_vtk_target})
                    sindrecpp_copy_runtime_dirs(${_sindrecpp_vtk_target}
                        "${_sindrecpp_vtk_runtime_dir}")
                endif()
            endforeach()
            if(SINDRECPP_UTILS3D_VTK_DATA)
                list(APPEND _sindrecpp_vtk_tests sindrecpp.vtk_data)
            endif()
            if(SINDRECPP_UTILS3D_SHOW)
                list(APPEND _sindrecpp_vtk_tests sindrecpp.show_mesh)
            endif()
            set_tests_properties(${_sindrecpp_vtk_tests} PROPERTIES
                ENVIRONMENT_MODIFICATION "${_sindrecpp_vtk_path_modification}")
        endif()
    endif()
endif()

if(SINDRECPP_BUILD_UTILS3D_BENCHMARKS)
    if(NOT TARGET BLAS::BLAS)
        message(WARNING "Utils3d benchmark requested, but BLAS::BLAS is unavailable")
    else()
        add_executable(sindrecpp_utils3d_benchmark tests/utils3d_benchmark.cpp)
        target_link_libraries(sindrecpp_utils3d_benchmark PRIVATE SindreCpp::Utils3d BLAS::BLAS)
        sindrecpp_copy_target_runtime_dlls(sindrecpp_utils3d_benchmark)
    endif()
endif()

install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
